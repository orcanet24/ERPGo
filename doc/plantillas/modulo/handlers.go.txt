// SUSTITUCIONES: hay DOS marcadores, y la diferencia es intencionada.
//
//	<MODULO>  minúscula. Nombres de tabla, ruta, directorio y plantilla.
//	<Modulo>  mayúscula. Tipos y funciones: DemomodInputs, calcularDemomod.
//	modulo    sin ángulos. Handlers: los métodos son NO exportados
//	          (func (s *Server) moduloList), porque el layout y las
//	          plantillas los invocan por ruta, no por nombre de función.
//
// Si sustituyes los tres con la misma cosa y el módulo se llama "ventas",
// los handlers quedan exported por accidente. No rompe nada, pero rompe la
// convención del resto del repo.
//
// POR QUÉ ESTE FICHERO ESTÁ SEPARADO DEL SERVICIO
//
// El handler es delgado a propósito: valida, delega, traduce a JSON o a
// plantilla. Toda la regla de negocio vive en <MODULO>.go, que es donde se
// prueba sin levantar un servidor HTTP. Referencia real del repo:
// crm_handlers.go (esto) contra crm_service.go (la lógica).
//
// CUATRE REGLAS QUE TODO HANDLER DEBE CUMPLIR
//
//  1. El método HTTP va EN LA LÍNEA de la ruta, no en un if del handler:
//         s.mux.HandleFunc("POST /x/save", s.xSave)
//     No "if r.Method == ..." ni rutas registradas con el método equivocado.
//
//  2. db SIEMPRE por s.dbFor(r), nunca s.db. dbFor resuelve la base de la
//     empresa de la sesión; s.db es la de la empresa activa y en un ERP
//     multiempresa eso es escritura cruzada entre clientes.
//
//  3. Todo POST pasa por csrfValid antes de tocar nada.
//
//  4. El error se responde con jsonError / writeJSON. El formato de la
//     respuesta de error es parte del contrato con el JavaScript de la
//     pantalla, y se responde siempre igual.

import (
	"net/http"
	"strconv"
	"strings"
)

// register<Modulo>Routes registra TODAS las rutas del módulo en un solo sitio.
//
// Y OJO: registrar la función NO la activa. Hay que llamar a esta función
// desde NewServer en server.go. Ese es el paso 3 del patrón y es el que más
// veces se ha olvidado: el módulo compila, pasa vet, pasa los tests, y no
// aparece porque nadie invocó la función.
func (s *Server) register<Modulo>Routes() {
	s.mux.HandleFunc("GET /<MODULO>", s.moduloList)
	s.mux.HandleFunc("GET /<MODULO>/data", s.moduloListData)
	s.mux.HandleFunc("GET /<MODULO>/add", s.moduloForm)
	s.mux.HandleFunc("GET /<MODULO>/view/{id}", s.moduloView)
	s.mux.HandleFunc("POST /<MODULO>/save", s.moduloSave)
}

// moduloList pinta la pantalla vacía. El JavaScript pide los datos aparte:
// es lo que hacen las 110 pantallas del repo y evita recargar la tabla entera.
func (s *Server) moduloList(w http.ResponseWriter, r *http.Request) {
	s.render(w, r, "<MODULO>.html", map[string]interface{}{
		"Title":  "<Módulo>",
		"Module": "<MODULO>",
	})
}

// moduloListData devuelve el JSON de la tabla.
func (s *Server) moduloListData(w http.ResponseWriter, r *http.Request) {
	periodo := r.URL.Query().Get("periodo")
	if periodo == "" {
		periodo = today()
	}

	// dbFor, no s.db: la empresa es la de la petición.
	filas, total, err := s.listar<Modulo>(s.dbFor(r), periodo)
	if err != nil {
		jsonError(w, err.Error(), 500)
		return
	}

	writeJSON(w, map[string]interface{}{
		"status":  1,
		"rows":    filas,
		"total":   total,
		"periodo": periodo,
	})
}

func (s *Server) moduloForm(w http.ResponseWriter, r *http.Request) {
	s.render(w, r, "<MODULO>_form.html", map[string]interface{}{
		"Title":  "Nuevo <Módulo>",
		"Module": "<MODULO>",
	})
}

// moduloSave escribe. Orden inviolable: CSRF, parsear, validar, calcular,
// guardar, responder. Si se invierte, se guarda lo que se pudo validar a medias.
func (s *Server) moduloSave(w http.ResponseWriter, r *http.Request) {
	if err := r.ParseForm(); err != nil {
		jsonError(w, "formulario ilegible", 400)
		return
	}
	if !s.csrfValid(r, s.sessionFromRequest(r)) {
		jsonError(w, "token de seguridad inválido: recarga la página", 403)
		return
	}

	code := strings.TrimSpace(r.FormValue("code"))
	desc := r.FormValue("description")
	// El texto viene del formulario, y un decimal no cabe en float64 sin
	// pasar por el helper: parseMoney acepta "1.234,56" y "1234.56".
	cantidad := parseMoney(r.FormValue("quantity"))
	monto := parseMoney(r.FormValue("amount"))

	// Validar ANTES de la base, con el mensaje en español que ve el usuario.
	if err := validar<Modulo>(code, desc, cantidad, monto); err != nil {
		jsonError(w, err.Error(), 400)
		return
	}

	// El cálculo puro, siempre. El handler no hace aritmética de negocio.
	//
	// Ojo: lo que se ESCRIBE es la cantidad que pidió el usuario, no un
	// campo inventado del resultado. El breakdown solo trae el total y el
	// unitario; si el cálculo dedujera una cantidad efectiva distinta (por
	// rendimiento, por merma), ESA es la que hay que escribir, y entonces
	// el desglose tiene que exponerla con ese nombre. Inventar un campo en
	// el handler es como empiezan los descuadres que nadie sabe explicar.
	out := calcular<Modulo>(<Modulo>Inputs{
		Quantity: cantidad,
		Rate:     monto,
	})

	db := s.dbFor(r)
	if _, err := db.Exec(`
		INSERT INTO <MODULO>_main (code, description, quantity, amount, period)
		VALUES (?, ?, ?, ?, CURDATE())`, code, desc, cantidad, out.Total); err != nil {
		jsonError(w, "no se pudo guardar: "+err.Error(), 500)
		return
	}

	writeJSON(w, map[string]interface{}{"status": 1, "msg": "Guardado", "total": out.Total})
}

func (s *Server) moduloView(w http.ResponseWriter, r *http.Request) {
	id, err := strconv.Atoi(r.PathValue("id"))
	if err != nil {
		jsonError(w, "identificador inválido", 400)
		return
	}
	s.render(w, r, "<MODULO>_view.html", map[string]interface{}{
		"Title":  "<Módulo>",
		"Module": "<MODULO>",
		"ID":     id,
	})
}
